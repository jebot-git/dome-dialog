cmake_minimum_required(VERSION 3.24)
project(dome-dialog LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(FLTK CONFIG REQUIRED HINTS /opt/fltk-vr/toolkit/share/fltk)
find_package(PkgConfig REQUIRED)
find_package(ZLIB REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
add_executable(dome-dialog main.cpp)
target_include_directories(dome-dialog PRIVATE ../flterm-xr/xr)
target_link_libraries(dome-dialog PRIVATE fltk::fltk-shared PkgConfig::GIO ZLIB::ZLIB)
set_target_properties(dome-dialog PROPERTIES INSTALL_RPATH /opt/fltk-vr/toolkit/lib)
install(TARGETS dome-dialog DESTINATION bin)
